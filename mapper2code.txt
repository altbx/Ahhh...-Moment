#!/usr/bin/env python

import sys
d={}
for line in sys.stdin:
        w,c=line.strip().split("\t")
        d[w]=d.get(w,0)+int(c)

for k,v in sorted(d.items(),key=lambda item:item[1],reverse=True):
        print("%s\t%d"%(k,v))
