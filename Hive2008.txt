CREATE DATABASE mohitdb;
USE mohitdb;

CREATE TABLE flight(
flightno INT,
dest VARCHAR(10),
year INT,
day INT,
delay FLOAT
);

ALTER TABLE flight RENAME TO flightn;

DROP TABLE flightn;

CREATE TABLE flight1(
flightno INT,
dest VARCHAR(10),
year INT,
day INT,
delay FLOAT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
LINES TERMINATED BY '\n'
STORED AS TEXTFILE;

INSERT INTO flight1 VALUES
(101,'pune',2008,1,30.01),
(102,'cat',2008,2,30.37);

CREATE EXTERNAL TABLE flightex(
flightno INT,
dest VARCHAR(10),
year INT,
day INT,
delay FLOAT
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE
LOCATION '/user/hive/flightex';

INSERT INTO flightex SELECT * FROM flight1;

SELECT * FROM flightex;

CREATE TABLE flightj(
flightno INT,
source VARCHAR(10)
)
ROW FORMAT DELIMITED
FIELDS TERMINATED BY ','
STORED AS TEXTFILE;

INSERT INTO flightj VALUES(101,'flih');

SELECT * 
FROM flight1 a
JOIN flightj b
ON a.flightno=b.flightno;

SELECT day,AVG(delay)
FROM flight1
WHERE year=2008
GROUP BY day;

CREATE INDEX flight_index
ON TABLE flight1(flightno)
AS 'org.apache.hadoop.hive.ql.index.compact.CompactIndexHandler'
WITH DEFERRED REBUILD;

ALTER INDEX flight_index ON flight1 REBUILD;
